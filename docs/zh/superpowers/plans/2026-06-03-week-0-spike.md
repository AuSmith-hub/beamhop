```swift
import Foundation

// Minimal MCP server over stdio (JSON-RPC 2.0). Implements:
//  - initialize
//  - tools/list -> returns beamhop_fetch_capture
//  - tools/call -> returns fixed payload regardless of args
// Throw-away spike code, no error handling polish.

struct JSONRPC: Codable {
    var jsonrpc: String = "2.0"
    var id: AnyCodableValue?
    var method: String?
    var params: AnyCodableValue?
    var result: AnyCodableValue?
    var error: AnyCodableValue?
}

enum AnyCodableValue: Codable {
    case string(String)
    case int(Int)
    case object([String: AnyCodableValue])
    case array([AnyCodableValue])
    case bool(Bool)
    case null

    func encode(to encoder: Encoder) throws {
        var c = encoder.singleValueContainer()
        switch self {
        case .string(let s): try c.encode(s)
        case .int(let i): try c.encode(i)
        case .object(let o): try c.encode(o)
        case .array(let a): try c.encode(a)
        case .bool(let b): try c.encode(b)
        case .null: try c.encodeNil()
        }
    }
    init(from decoder: Decoder) throws {
        let c = try decoder.singleValueContainer()
        if c.decodeNil() { self = .null; return }
        if let b = try? c.decode(Bool.self) { self = .bool(b); return }
        if let i = try? c.decode(Int.self) { self = .int(i); return }
        if let s = try? c.decode(String.self) { self = .string(s); return }
        if let o = try? c.decode([String: AnyCodableValue].self) { self = .object(o); return }
        if let a = try? c.decode([AnyCodableValue].self) { self = .array(a); return }
        self = .null
    }
}

let stdin = FileHandle.standardInput
let stdout = FileHandle.standardOutput
let stderr = FileHandle.standardError

func log(_ s: String) {
    stderr.write(Data("[hello-server] \(s)\n".utf8))
}

func send(_ obj: [String: AnyCodableValue]) {
    let data = try! JSONEncoder().encode(obj)
    let header = "Content-Length: \(data.count)\r\n\r\n"
    stdout.write(Data(header.utf8))
    stdout.write(data)
}

func handle(_ msg: [String: AnyCodableValue]) {
    guard case let .string(method)? = msg["method"] else { return }
    let id = msg["id"] ?? .null
    log("got method=\(method)")
    switch method {
    case "initialize":
        send([
            "jsonrpc": .string("2.0"),
            "id": id,
            "result": .object([
                "protocolVersion": .string("2024-11-05"),
                "capabilities": .object(["tools": .object([:])]),
                "serverInfo": .object(["name": .string("beamhop-hello"), "version": .string("0.0.1")])
            ])
        ])
    case "tools/list":
        send([
            "jsonrpc": .string("2.0"),
            "id": id,
            "result": .object([
                "tools": .array([
                    .object([
                        "name": .string("beamhop_fetch_capture"),
                        "description": .string("Fetch a Beamhop capture by id (spike: returns fixed payload)"),
                        "inputSchema": .object([
                            "type": .string("object"),
                            "properties": .object([
                                "id": .object(["type": .string("string")])
                            ])
                        ])
                    ])
                ])
            ])
        ])
    case "tools/call":
        send([
            "jsonrpc": .string("2.0"),
            "id": id,
            "result": .object([
                "content": .array([
                    .object([
                        "type": .string("text"),
                        "text": .string("SPIKE_S1_OK: this is the fixed capture payload from hello-server.")
                    ])
                ])
            ])
        ])
    default:
        send([
            "jsonrpc": .string("2.0"),
            "id": id,
            "error": .object(["code": .int(-32601), "message": .string("method not found")])
        ])
    }
}

// Read Content-Length framed JSON-RPC messages
var buffer = Data()
while let chunk = try? stdin.read(upToCount: 4096), !chunk.isEmpty {
    buffer.append(chunk)
    while let hdrEnd = buffer.range(of: Data("\r\n\r\n".utf8)) {
        let header = String(data: buffer.subdata(in: 0..<hdrEnd.lowerBound), encoding: .utf8) ?? ""
        guard let lenLine = header.split(separator: "\r\n").first(where: { $0.starts(with: "Content-Length:") }),
              let len = Int(lenLine.split(separator: ":").last?.trimmingCharacters(in: .whitespaces) ?? "")
        else { buffer.removeSubrange(0..<hdrEnd.upperBound); continue }
        let bodyStart = hdrEnd.upperBound
        guard buffer.count >= bodyStart + len else { break }
        let body = buffer.subdata(in: bodyStart..<bodyStart+len)
        buffer.removeSubrange(0..<bodyStart+len)
        if let json = try? JSONDecoder().decode([String: AnyCodableValue].self, from: body) {
            handle(json)
        }
    }
}
```

```javascript
let port = null;
function ensurePort() {
  if (port) return port;
  port = chrome.runtime.connectNative('com.beamhop.bridge');
  port.onDisconnect.addListener(() => { port = null; console.warn('host disconnected'); });
  return port;
}

const pending = new Map();
let seq = 0;

chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  const id = ++seq;
  const p = ensurePort();
  const handler = (resp) => {
    if (resp.id === id) { p.onMessage.removeListener(handler); sendResponse(resp); }
  };
  p.onMessage.addListener(handler);
  p.postMessage({ id, ...msg });
  return true; // keep sendResponse alive
});
```

- [ ] **Step 2.5: Write native host in Swift**

`spike/s4-chrome-native-messaging/host/main.swift`:

```swift
import Foundation

let stdin = FileHandle.standardInput
let stdout = FileHandle.standardOutput
let stderr = FileHandle.standardError

func log(_ s: String) { stderr.write(Data("[bridge] \(s)\n".utf8)) }

while true {
    // Chrome native messaging frame: 4-byte little-endian length prefix + JSON
    guard let lenData = try? stdin.read(upToCount: 4), lenData.count == 4 else { break }
    let len = lenData.withUnsafeBytes { $0.load(as: UInt32.self).littleEndian }
    guard len > 0 && len < 16 * 1024 * 1024 else { log("bad length \(len)"); break }
    guard let body = try? stdin.read(upToCount: Int(len)), body.count == Int(len) else { break }

    guard let obj = try? JSONSerialization.jsonObject(with: body) as? [String: Any] else {
        log("bad json"); continue
    }
    let id = obj["id"] as? Int ?? 0
    let kind = obj["kind"] as? String ?? ""

    var resp: [String: Any] = ["id": id]
    switch kind {
    case "ping":
        resp["pong"] = true
        resp["host_version"] = "0.0.1"
    case "echo":
        resp["body"] = obj["body"] ?? ""
    default:
        resp["error"] = "unknown kind"
    }

    let respData = try! JSONSerialization.data(withJSONObject: resp)
    var lenLE = UInt32(respData.count).littleEndian
    stdout.write(Data(bytes: &lenLE, count: 4))
    stdout.write(respData)
}
```

```swift
import Cocoa
import ApplicationServices

// Run from a Terminal that already has Accessibility permission (System Settings → Privacy → Accessibility).
// Triggers on each Cmd+Shift+P press; logs JSON line per probe to stdout.

func currentFrontApp() -> (bundleID: String, name: String, pid: pid_t)? {
    guard let app = NSWorkspace.shared.frontmostApplication else { return nil }
    return (app.bundleIdentifier ?? "unknown", app.localizedName ?? "unknown", app.processIdentifier)
}

func axAttr(_ el: AXUIElement, _ key: String) -> Any? {
    var v: CFTypeRef?
    let r = AXUIElementCopyAttributeValue(el, key as CFString, &v)
    return r == .success ? v as Any : nil
}

func focusedElement(forPID pid: pid_t) -> AXUIElement? {
    let app = AXUIElementCreateApplication(pid)
    return axAttr(app, kAXFocusedUIElementAttribute as String) as! AXUIElement?
}

func probe() -> [String: Any] {
    guard let front = currentFrontApp() else { return ["error": "no front app"] }
    var result: [String: Any] = [
        "ts": ISO8601DateFormatter().string(from: Date()),
        "bundle_id": front.bundleID,
        "app_name": front.name,
        "pid": Int(front.pid),
    ]
    let appEl = AXUIElementCreateApplication(front.pid)
    if let win = axAttr(appEl, kAXFocusedWindowAttribute as String) {
        let winEl = win as! AXUIElement
        result["window_title"] = axAttr(winEl, kAXTitleAttribute as String) as? String ?? ""

        }
    }

let tap = CGEvent.tapCreate(tap: .cgSessionEventTap, place: .headInsertEventTap, options: .listenOnly,
                            eventsOfInterest: mask, callback: { _, _, event, _ in
    let keycode = event.getIntegerValueField(.keyboardEventKeycode)
    let flags = event.flags
    // Cmd (0x100000) + Shift (0x20000) + P (35)
    if keycode == 35 && flags.contains(.maskCommand) && flags.contains(.maskShift) {
        let r = probe()
        if let d = try? JSONSerialization.data(withJSONObject: r), let s = String(data: d, encoding: .utf8) {
            print(s)
            fflush(stdout)
        }
    }
    return Unmanaged.passRetained(event)
}, userInfo: nil)
```

```swift
class AppDelegate: NSObject, NSApplicationDelegate {

        NSApp.setActivationPolicy(.accessory)

        statusItem.button?.title = "✦"

        let menu = NSMenu()
        menu.addItem(NSMenuItem(title: "Show overlay", action: #selector(toggle), keyEquivalent: ""))
        menu.addItem(NSMenuItem(title: "Quit", action: #selector(quit), keyEquivalent: "q"))
        statusItem.menu = menu

        overlay = NSWindow(
            contentRect: NSRect(x: 0, y: 0, width: 480, height: 280),
            styleMask: [.titled, .closable],
            backing: .buffered, defer: false
        )
        overlay.title = "Beamhop floating demo"
        overlay.level = .floating
        overlay.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary]
        overlay.isMovableByWindowBackground = true
        overlay.contentView = NSHostingView(rootView: OverlayView())
        overlay.center()

        let mods: UInt32 = UInt32(cmdKey | shiftKey)
        let key: UInt32 = 49 // space

        hotKeyRef = ref
        var spec = EventTypeSpec(eventClass: OSType(kEventClassKeyboard), eventKind: OSType(kEventHotKeyPressed))
        InstallEventHandler(GetApplicationEventTarget(), { _, _, _ in
            DispatchQueue.main.async { (NSApp.delegate as? AppDelegate)?.toggle() }
            return noErr
        }, 1, &spec, nil, nil)
    }

    @objc func toggle() {
        if overlay.isVisible { overlay.orderOut(nil) }
        else { overlay.makeKeyAndOrderFront(nil); NSApp.activate(ignoringOtherApps: true) }
    }
    @objc func quit() { NSApp.terminate(nil) }
}
```
